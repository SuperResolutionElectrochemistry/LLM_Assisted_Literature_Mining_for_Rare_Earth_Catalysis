# 4. GPT调用
# =========================

def call_gpt(title, abstract):
    try:
        res = client.chat.completions.create(
            model=model_name,
            messages=[
                {"role": "system", "content": SYSTEM_PROMPT},
                {"role": "user", "content": USER_PROMPT.format(title=title, abstract=abstract)}
            ],
            temperature=0,
            timeout=60
        )
        return res.choices[0].message.content.strip()
    except Exception as e:
        return "API_Error"

# =========================
# 5. 解析函数（生成Sheet2关键）
# =========================

def parse_edges(ref_id, raw_text):
    rows = []

    if raw_text in ("NULL", "API_Error"):
        return rows

    lines = raw_text.split("\n")

    for line in lines:
        if "->" not in line:
            continue

        try:
            parts = [x.strip() for x in line.split("->")]
            if len(parts) < 5:
                continue

            element = parts[0]
            dimension = parts[1]
            descriptor = parts[2]
            relation = parts[3]
            rest = "->".join(parts[4:])

            rows.append({
                "Reference Number": ref_id,
                "Rare Earth Element": element,
                "Design Dimension": dimension,
                "Descriptor/Mechanism": descriptor,
                "Relation": relation,
                "Evidence Sentence + Confidence": rest
            })
        except:
            continue

    return rows
